# One combined venv for the whole pipeline (sample generation + training).
# No separate UI/STT/auto-training stack, no separate venv per Python version.

# --- sample generation (piper-sample-generator) ---
# Pinned to a pre-2.10 release: torch>=2.10 switched its bundled CUDA
# libraries from the nvidia-*-cu12 packages to nvidia-*-cu13, which do NOT
# overlap with tensorflow[and-cuda]'s cu12 requirement below. Staying under
# 2.10 means pip installs ONE shared set of cu12 libraries for both
# frameworks instead of two (saves ~2-3GB and avoids two CUDA runtimes
# fighting over the GPU in the same venv).
torch<2.10,>=2.9
torchaudio
# `datasets` (below) needs torchcodec to encode/decode Audio-feature columns
# (used by 01_fetch_assets.py streaming MIT RIR / AudioSet from HF). Version-
# locked to torch like torchaudio -- torchcodec's own compatibility table
# (github.com/meta-pytorch/torchcodec) pairs 0.9.x with torch 2.9.x, matching
# our pin above. Bump both together if the torch pin ever moves.
torchcodec<0.10,>=0.9
# NOT `pip install piper-sample-generator`: its PyPI wheel only bundles the
# piper_sample_generator package (see its pyproject.toml:
# packages.find.include = ["piper_sample_generator*"]), but __main__.py's
# .pt-checkpoint code path imports the sibling `piper_train` package, which
# lives in the same GitHub repo but was never published -- there's no
# `piper-train` on PyPI at all. Upstream's own notebook works around this by
# git-cloning the whole repo and adding it to sys.path instead of pip
# installing; 01_fetch_assets.py does the same (see fetch_piper_sample_generator_src
# and PIPER_SAMPLE_GENERATOR_REF), and 02_generate_samples.py points
# PYTHONPATH at that clone. Just need its actual runtime deps here:
piper-tts==1.3.0
# NOT plain `webrtcvad`: piper-sample-generator's own pyproject.toml lists it
# (unpinned) but __main__.py's .pt-checkpoint code path never actually
# imports it, and microwakeword needs a `webrtcvad` module too (via
# webrtcvad-wheels below) -- installing both means whichever installs last
# silently overwrites the other's files in site-packages, since both provide
# the same `webrtcvad` import name. That's not just messy: confirmed for
# real that plain `webrtcvad` won that race and broke the whole pipeline --
# its bundled webrtcvad.py does `import pkg_resources` for version lookup on
# Python <3.8, which recent setuptools doesn't reliably bundle anymore.
# webrtcvad-wheels' webrtcvad.py uses importlib.metadata on 3.8+ instead
# (verified by inspecting its wheel directly) and provides the exact same
# `webrtcvad` module/API, so it alone covers everything that needs it.

# --- microWakeWord training (not on PyPI, installed straight from git) ---
# Must be -e (editable), not a normal `name @ git+URL` install: microwakeword's
# setup.py uses packages=find_packages(), and microwakeword/audio/ has no
# __init__.py, so a real wheel build silently drops that whole subpackage
# (confirmed empirically -- `import microwakeword.audio.augmentation` raises
# ModuleNotFoundError after a plain git install, no error at install time).
# An editable install clones into the venv's src/ and imports straight from
# that live tree via Python's implicit namespace packages, which sidesteps
# find_packages() entirely since nothing gets copied into a wheel manifest.
-e git+https://github.com/kahrendt/microWakeWord.git#egg=microwakeword
# >=0.35.0: microwakeword's Augmentation class uses AddColorNoise, added in
# that release. Needs an actual floor, not just "audiomentations" unpinned --
# pip won't upgrade an already-installed version that satisfies an unpinned
# requirement line, so a stale older version (e.g. pulled in transitively by
# something else's hard pin earlier) silently sticks around across reinstalls
# even after that something else is removed. Confirmed hitting exactly this:
# 0.33.0 (piper-sample-generator's old pin, before we stopped pip-installing
# it) lacks AddColorNoise entirely.
audiomentations>=0.35.0
audio_metadata
datasets
mmap_ninja
numpy>=2.0
pymicro-features
pyyaml
tensorflow[and-cuda]>=2.18
